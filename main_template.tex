\newcommand*{\TryRefsPath}[1]{%
    \ifx\RefsPath\empty
        \IfFileExists{#1/\RefsFilename.bib}{%
            \gdef\RefsPath{#1}%
        }{}%
    \fi
}%

% A content frame holds prose, bullets and equations in any order.
\begin{contentframe}[title=Introduction, subtitle=A Smart subtitle,
                     label=slide:introduction]

    \citet[p. 385]{bib:becker1993}:

    \begin{displayquote}
        ``The economic approach is a method of analysis, not an assumption of
        particular motivations. [\ldots] The analysis assumes that individuals
        maximize welfare as they conceive it, whether they be selfish,
        altruistic, loyal, spiteful, or masochistic.''
    \end{displayquote}

    \begin{itemize}
        \item A bullet, with a link that belongs to this claim:
              \slidelink{slide:backup-detail}{the derivation}.
        \item[$\rightarrow$] Custom markers work as usual.
    \end{itemize}

\end{contentframe}

% A standout frame carries one sentence, full bleed, palette inverted.
\begin{standoutframe}
    The one sentence the room should leave with.
\end{standoutframe}

% A figure frame holds exactly one \slidefigure, centred and fitted.
\begin{figureframe}[title=A Figure, subtitle=What it shows,
                    label=slide:figure]
    \slidefigure[width=.7\linewidth,
                 caption={A caption in the template's house style: the long
                          sentence that says what the reader is looking at.},
                 note={Notes: a short source or reading line.},
                 label=fig:example]{example-image}
\end{figureframe}

% A table frame holds exactly one \slidetable. By default the path names a .tex
% fragment the estimation code wrote; kind=file draws an image instead.
\begin{tableframe}[title=A Table, subtitle=Estimates, label=slide:table]
    \slidetable[width=.7\linewidth,
                caption={Difference-in-differences estimates of the outcome on
                         treatment, with and without controls.},
                note={Standard errors in parentheses.},
                label=tab:example]{\TablesPath/example_estimates}
\end{tableframe}

% A split frame is two columns: write the left one, then \splitnext, then the
% right one. Text beside a figure and two columns of bullets are the same frame.

    \end{itemize}

    \splitnext

    \slidefigure[width=\linewidth, caption={The evidence on the right.}]%
                {example-image}

\end{splitframe}
